local record M
   n: integer
end

-- A block whose body is nothing but a comment. The parser gives such a body no extent of
-- its own -- its `statements` node starts and ends at the `end` that closes it -- so
-- before #231 every one of these lost a level and kept it, the output being a fixed point.
local function empties(x: integer): integer
   if x > 0 then
      -- misindented on purpose: the formatter puts it back
   end
   if x > 0 then
      return 1
   elseif x < 0 then
      -- an elseif body
   else
      -- and an else body
   end
   while x > 100 do
      -- a while body
   end
   for i = 1, 3 do
      -- a numeric for body
   end
   for _, v in ipairs({ 1, 2 }) do
      -- a generic for body
   end
   repeat
      -- a repeat body, which ends at `until` rather than `end`
   until true
   do
      -- a plain do block
   end
   return x
end

-- A body that does have statements, whose first line is a comment. The `statements` node
-- starts at the first *statement*, so the comment above it was outside the block.
local function first_line(x: integer): integer
   -- what this function is for
   local y = x + 1
   if y > 3 then
      -- why this branch exists
      return y
   end
   return 0
end

-- The same, one level in: a method, and a function value inside it.
function M:method(): integer
   -- the method's first line
   local f = function(y: integer): integer
      -- the nested function's first line
      return y + 1
   end
   return f(self.n)
end

-- A comment-only table constructor was always right: `literal_table`'s extent is the
-- braces, not its contents. It is here so that staying right is checked too.
local empty: {string:integer} = {
   -- nothing in here yet
}

-- A wrapped condition keeps its own shape: the body opens at `then`, so the lines of the
-- condition are not part of it.
local function wrapped(x: integer): integer
   if x > 0
      and x < 10 then
      -- inside the body, not inside the condition
      return 1
   end
   return 0
end

-- A function whose header spans lines: the body opens after the return type.
local function header(
   x: integer,
   y: integer
): integer
   -- the first line of a body under a wrapped header
   return x + y
end

return { M = M, empties = empties, first_line = first_line, wrapped = wrapped, header = header, empty = empty }
